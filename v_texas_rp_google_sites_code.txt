      // ==================== INDEXEDDB ENGINE ====================
      function openIndexedDB() {
        return new Promise((resolve) => {
          if (!window.indexedDB) {
            resolve(null);
            return;
          }
          try {
            const req = window.indexedDB.open(IDB_NAME, IDB_VERSION);
            req.onupgradeneeded = function (e) {
              const db = e.target.result;
              if (!db.objectStoreNames.contains(IDB_STORE)) {
                db.createObjectStore(IDB_STORE, { keyPath: 'id' });
              }
            };
            req.onsuccess = function (e) {
              dbInstance = e.target.result;
              resolve(dbInstance);
            };
            req.onerror = function () {
              resolve(null);
            };
          } catch (e) {
            resolve(null);
          }
        });
      }

      function idbGetAll() {
        return new Promise((resolve) => {
          if (!dbInstance) {
            resolve([]);
            return;
          }
          try {
            const tx = dbInstance.transaction(IDB_STORE, 'readonly');
            const store = tx.objectStore(IDB_STORE);
            const req = store.getAll();
            req.onsuccess = () => resolve(req.result || []);
            req.onerror = () => resolve([]);
          } catch (e) {
            resolve([]);
          }
        });
      }

      function idbPut(item) {
        return new Promise((resolve) => {
          if (!dbInstance) {
            resolve(false);
            return;
          }
          try {
            const tx = dbInstance.transaction(IDB_STORE, 'readwrite');
            const store = tx.objectStore(IDB_STORE);
            const req = store.put(item);
            req.onsuccess = () => resolve(true);
            req.onerror = () => resolve(false);
          } catch (e) {
            resolve(false);
          }
        });
      }

      function idbPutAll(items) {
        return new Promise((resolve) => {
          if (!dbInstance) {
            resolve(false);
            return;
          }
          try {
            const tx = dbInstance.transaction(IDB_STORE, 'readwrite');
            const store = tx.objectStore(IDB_STORE);
            store.clear();
            items.forEach(item => {
              if (item && item.id) store.put(item);
            });
            tx.oncomplete = () => resolve(true);
            tx.onerror = () => resolve(false);
          } catch (e) {
            resolve(false);
          }
        });
      }

      function normalizeCloudUrl(url) {
        if (!url) return '';
        let clean = url.trim().replace(/\/+$/, '');
        if (clean.endsWith('.json')) {
          clean = clean.substring(0, clean.lastIndexOf('/'));
        }
        if (clean.endsWith('/applications')) {
          clean = clean.substring(0, clean.lastIndexOf('/applications'));
        }
        return clean;

      async function fetchCloudApplications() {
        const base = normalizeCloudUrl(getCloudDbUrl());
        if (!base) return null;
        try {
          const res = await fetch(`${base}/applications.json`, { cache: 'no-store' });
          if (!res.ok) throw new Error(`HTTP ${res.status}`);
          const data = await res.json();
          if (!data) return [];
          if (Array.isArray(data)) {
            return data.filter(item => item && item.id);
          }
          if (typeof data === 'object') {
            return Object.values(data).filter(item => item && item.id);
          }
          return [];
        } catch (err) {
          console.warn('Cloud DB fetch error:', err);
          throw err;
        }
      }

      async function pushApplicationToCloud(app) {
        const base = normalizeCloudUrl(getCloudDbUrl());
        if (!base || !app || !app.id) return false;
        try {
          const res = await fetch(`${base}/applications/${encodeURIComponent(app.id)}.json`, {
            method: 'PUT',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify(app)
          });
          return res.ok;
        } catch (e) {
          console.warn('Cloud push failed:', e);
          return false;
        }
      }

      async function patchApplicationInCloud(appId, patch) {
        const base = normalizeCloudUrl(getCloudDbUrl());
        if (!base || !appId) return false;
        try {
          const res = await fetch(`${base}/applications/${encodeURIComponent(appId)}.json`, {
            method: 'PATCH',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify(patch)
          });
          return res.ok;
        } catch (e) {
          console.warn('Cloud patch failed:', e);
          return false;
        }
      }

      async function syncWithCloud(silent) {
        const cloudUrl = getCloudDbUrl();
        if (!cloudUrl) {
          if (!silent) showToast('لم يتم ضبط رابط قاعدة البيانات السحابية بعد.', 'error');
          updateCloudStatusUI();
          return;
        }

        const syncBtn = document.getElementById('syncCloudDbBtn');
        const syncIcon = document.getElementById('syncIcon');
        if (syncIcon) syncIcon.classList.add('animate-spin');
        if (syncBtn) syncBtn.classList.add('opacity-75');

        try {
          const cloudApps = await fetchCloudApplications();
          if (cloudApps === null) throw new Error('تعذر جلب البيانات');

          const appMap = new Map();
          applications.forEach(a => {
            if (a && a.id) appMap.set(a.id, a);
          });

          cloudApps.forEach(cApp => {
            if (!cApp || !cApp.id) return;
            if (!appMap.has(cApp.id)) {
              appMap.set(cApp.id, cApp);
            } else {
              const local = appMap.get(cApp.id);
              if (cApp.status !== local.status || (cApp.adminNote && cApp.adminNote !== local.adminNote)) {
                appMap.set(cApp.id, Object.assign({}, local, cApp));
              }
            }
          });

          const cloudIds = new Set(cloudApps.map(c => c && c.id));
          const uploadQueue = [];
          applications.forEach(localApp => {
            if (localApp && localApp.id && !cloudIds.has(localApp.id)) {
              uploadQueue.push(localApp);
            }
          });

          applications = Array.from(appMap.values());
          applications.sort((a, b) => (b.createdAt || '').localeCompare(a.createdAt || ''));

          saveState();
          updateCounters();
          renderAdminDashboard();
          updateCloudStatusUI();

          if (uploadQueue.length > 0) {
            Promise.all(uploadQueue.map(a => pushApplicationToCloud(a))).catch(() => {});

          if (!silent) {
            showToast(`🟢 تمت المزامنة السحابية بنجاح! تم فحص وتحديث ${applications.length} طلب.`, 'info');
          }
        } catch (err) {
          console.error('Sync failed:', err);
          if (!silent) {
            showToast('❌ تعذر إتمام المزامنة السحابية. تأكد من صحة الرابط وقواعد Firebase.', 'error');
          }
        } finally {
          if (syncIcon) syncIcon.classList.remove('animate-spin');
          if (syncBtn) syncBtn.classList.remove('opacity-75');
        }

          const s = parseInt(btn.getAttribute('data-step-btn'), 10);
          const badge = btn.querySelector('span');
          if (s === step) {
            btn.className = 'step-indicator flex items-center gap-2 px-3 sm:px-4 py-2 rounded-xl text-xs sm:text-sm font-bold bg-amber-500 text-black shadow-lg';
            if (badge) badge.className = 'w-5 h-5 rounded-full bg-black text-amber-400 flex items-center justify-center text-xs font-black';
          } else if (s < step) {
            btn.className = 'step-indicator flex items-center gap-2 px-3 sm:px-4 py-2 rounded-xl text-xs sm:text-sm font-bold bg-amber-500/20 text-amber-300 border border-amber-500/30';
            if (badge) badge.className = 'w-5 h-5 rounded-full bg-amber-500/30 text-amber-300 flex items-center justify-center text-xs font-black';
          } else {
            btn.className = 'step-indicator flex items-center gap-2 px-3 sm:px-4 py-2 rounded-xl text-xs sm:text-sm font-bold bg-gray-900 text-gray-500 border border-gray-800';
            if (badge) badge.className = 'w-5 h-5 rounded-full bg-gray-800 text-gray-400 flex items-center justify-center text-xs font-black';
          }
        });
        window.scrollTo({ top: 120, behavior: 'smooth' });
      }

        let found = applications.find(
          a => a.id.toLowerCase() === q ||
               a.discordTag.toLowerCase() === q ||
               a.discordId.toLowerCase() === q
        );

        if (!found && getCloudDbUrl()) {
          container.innerHTML = `
            <div class="p-8 rounded-2xl bg-gray-900/50 border border-gray-800 text-center space-y-3">

        if (applications.length === 0) {
          showToast('لا توجد أي طلبات لتصديرها حالياً.', 'error');
          return;
        }
        const headers = [
          'رقم التقديم',
          'الاسم الكامل',
          'العمر',
          'اسم الشخصية (RP)',
          'يوزر الديسكورد',

          'المايك',

          'الحالة الإدارية',
          'تاريخ التقديم',
          'ملاحظات الإدارة'
        ];

        const escapeCsv = (val) => {
          if (val === null || val === undefined) return '""';
          const s = String(val).replace(/"/g, '""');
          return `"${s}"`;
        };

        const rows = applications.map(a => [

          escapeCsv(a.name),
          escapeCsv(a.age),
          escapeCsv(a.charName),
          escapeCsv(a.discordTag),
          escapeCsv(a.discordId),
          escapeCsv(a.country),
          escapeCsv(a.department),
          escapeCsv(a.hours),
          escapeCsv(a.shift),
          escapeCsv(a.mic),
          escapeCsv(a.quizScore + '%'),
          escapeCsv(a.status),
          escapeCsv(a.createdAt),
          escapeCsv(a.adminNote)
        ].join(','));

        const csvContent = '\uFEFF' + headers.join(',') + '\r\n' + rows.join('\r\n');
        const blob = new Blob([csvContent], { type: 'text/csv;charset=utf-8;' });
        const url = URL.createObjectURL(blob);
        const dlAnchor = document.createElement('a');
        dlAnchor.href = url;
        dlAnchor.download = `v_texas_rp_applications_${new Date().toISOString().slice(0, 10)}.csv`;
        document.body.appendChild(dlAnchor);
        dlAnchor.click();
        dlAnchor.remove();
        URL.revokeObjectURL(url);
        showToast('تم تصدير ملف Excel (CSV) بنجاح!', 'info');
      }

      function handleImportFile(file) {
        if (!file) return;
        const reader = new FileReader();
        reader.onload = function (e) {
          try {
            const data = JSON.parse(e.target.result);
            if (!Array.isArray(data)) {
              showToast('ملف غير صالح: يجب أن يحتوي الملف على مصفوفة طلبات JSON.', 'error');
              return;
            }
            const validApps = data.filter(item => item && item.id && item.name);
            if (validApps.length === 0) {
              showToast('لم يتم العثور على أي طلبات صالحة في الملف المرفق.', 'error');
              return;
            }

            pendingImportData = validApps;
            document.getElementById('importSummaryText').textContent = `تم فحص الملف بنجاح والعثور على ${validApps.length} طلب إداري جاهز للدمج.`;
            const modal = document.getElementById('importConfirmModal');
            modal.classList.remove('hidden');
            modal.classList.add('flex');
          } catch (err) {
            showToast('خطأ أثناء قراءة ملف JSON: تأكد من صحة الملف.', 'error');
          }
        };
        reader.readAsText(file);
      }

          const autoSync = document.getElementById('cloudAutoSyncCheckbox').checked;

          setCloudDbUrl(url);
          setCloudAutoSync(autoSync);

          const modal = document.getElementById('cloudDbModal');
          modal.classList.add('hidden');
          modal.classList.remove('flex');

          updateCloudStatusUI();

          if (url) {
            showToast('تم حفظ إعدادات قاعدة البيانات السحابية بنجاح! جاري المزامنة...', 'info');
            await syncWithCloud(false);
          } else {
            showToast('تم حفظ الإعدادات والاعتماد على التخزين المحلي الدائم.', 'info');
          }
        });

        document.getElementById('syncCloudDbBtn').addEventListener('click', () => {
          syncWithCloud(false);
        });

        // Export JSON
        document.getElementById('exportAppsJsonBtn').addEventListener('click', () => {
          const dataStr = 'data:text/json;charset=utf-8,' + encodeURIComponent(JSON.stringify(applications, null, 2));
          const dlAnchor = document.createElement('a');
          dlAnchor.setAttribute('href', dataStr);
          dlAnchor.setAttribute('download', `v_texas_rp_applications_${new Date().toISOString().slice(0, 10)}.json`);
          document.body.appendChild(dlAnchor);
          dlAnchor.click();
          dlAnchor.remove();
          showToast('تم تصدير ملف النسخة الاحتياطية (JSON) بنجاح.', 'info');
        });

        // Export CSV
        document.getElementById('exportAppsCsvBtn').addEventListener('click', () => {
          exportDatabaseCsv();
        });

        // Import JSON
        document.getElementById('importAppsJsonTriggerBtn').addEventListener('click', () => {
          document.getElementById('importJsonFileInput').click();
        });

        document.getElementById('importJsonFileInput').addEventListener('change', (e) => {
          const file = e.target.files && e.target.files[0];
          if (file) {
            handleImportFile(file);
            e.target.value = '';
          }
        });

        document.getElementById('cancelImportBtn').addEventListener('click', () => {
          pendingImportData = null;
          const modal = document.getElementById('importConfirmModal');
          modal.classList.add('hidden');
          modal.classList.remove('flex');
        });

        document.getElementById('executeImportBtn').addEventListener('click', () => {
          if (!pendingImportData) return;
          const map = new Map();
          applications.forEach(a => { if (a && a.id) map.set(a.id, a); });
          pendingImportData.forEach(a => { if (a && a.id) map.set(a.id, a); });
          applications = Array.from(map.values());
          applications.sort((a, b) => (b.createdAt || '').localeCompare(a.createdAt || ''));

          saveState();
          updateCounters();
          renderAdminDashboard();
          updateCloudStatusUI();

          const count = pendingImportData.length;
          pendingImportData = null;
          const modal = document.getElementById('importConfirmModal');
          modal.classList.add('hidden');
          modal.classList.remove('flex');

          showToast(`✅ تم استيراد ودمج ${count} طلب بنجاح في قاعدة البيانات!`, 'info');

          if (getCloudDbUrl()) {
            syncWithCloud(true);
          }
        });
      }

      loadState();
      updateCounters();
      bindEvents();
    })();
